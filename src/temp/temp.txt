@Slf4j
public class LocalWireMockRunner {

    private static final int PORT = 9000;
    private static final Path SOURCE_PDF = Path.of("C:", "Users", "S1626079", "temp", "umowa.pdf");
    private static final String ORIGINAL_FILE_NAME = "umowa.pdf";
    private static final int PAGE_COUNT = 6;
    
    // Różne czasy opóźnień dla symulacji
    private static final int SLOW_DELAY_MS = 20_000;
    private static final int FAST_DELAY_MS = 1_000;

    public static void main(String[] args) throws Exception {
        WireMockServer server = new WireMockServer(
                WireMockConfiguration.options()
                        .port(PORT)
                        .globalTemplating(true)
        );
        server.start();
        log.info("[WIREMOCK] started on port {} (baseUrl={})", PORT, server.baseUrl());

        registerStubs(server);

        CountDownLatch shutdownLatch = new CountDownLatch(1);
        Runtime.getRuntime().addShutdownHook(new Thread(() -> {
            log.info("[WIREMOCK] stopping");
            server.stop();
            shutdownLatch.countDown();
        }));

        log.info("[WIREMOCK] ready. Ctrl+C to stop.");
        shutdownLatch.await();
    }

    private static void registerStubs(WireMockServer server) throws Exception {
        if (!Files.exists(SOURCE_PDF)) {
            log.warn("[LOCAL-STUB] {} does not exist, skipping stub registration", SOURCE_PDF);
            return;
        }

        ObjectMapper objectMapper = new ObjectMapper();

        byte[] pdfBytes = Files.readAllBytes(SOURCE_PDF);
        String base64Pdf = java.util.Base64.getEncoder().encodeToString(pdfBytes);
        log.info("[LOCAL-STUB] loaded {} bytes from {} ({} chars base64)",
                pdfBytes.length, SOURCE_PDF, base64Pdf.length());

        List<String> pageTexts = IntStream.rangeClosed(1, PAGE_COUNT)
                .mapToObj(i -> "Zawartosc strony " + i + ".")
                .toList();

        // Dummy response
        String downloadJson = objectMapper.writeValueAsString(Map.of(
                "ocr", Map.of(ORIGINAL_FILE_NAME, pageTexts),
                "pdf", Map.of(ORIGINAL_FILE_NAME, List.of(base64Pdf))
        ));

        String inProgressJson = "{\"state\": \"IN PROGRESS\", \"finished\": false}";
        String doneJson = "{\"state\": \"SUCCESS\", \"finished\": true}";

        // 1. UPLOAD PDF -> Zawsze zwraca 200 OK
        server.stubFor(post(urlPathEqualTo("/api/uploadPdf"))
                .withHeader("Authorization", matching(".*"))
                .willReturn(aResponse()
                        .withStatus(200)
                        .withHeader("Content-Type", "application/json")
                        .withBody("{\"message\":\"PDF uploaded to Redis successfully\"}")));

        // 2. PROCESS PDF -> Rozdzielenie na szybki i wolny (WireMock Scenario)
        // Pierwszy plik wchodzący na taśmę dostanie ID "wolny-ocr-id"
        server.stubFor(post(urlPathEqualTo("/api/process"))
                .inScenario("Process-Speed-Scenario")
                .whenScenarioStateIs(Scenario.STARTED)
                .withHeader("Authorization", matching(".*"))
                .willReturn(aResponse()
                        .withStatus(200)
                        .withHeader("Content-Type", "application/json")
                        .withBody("{\"result_id\":\"wolny-ocr-id\"}"))
                .willSetStateTo("Wolny-Assigned"));

        // Każdy kolejny plik wchodzący na taśmę dostanie ID "szybki-ocr-id"
        server.stubFor(post(urlPathEqualTo("/api/process"))
                .inScenario("Process-Speed-Scenario")
                .whenScenarioStateIs("Wolny-Assigned")
                .withHeader("Authorization", matching(".*"))
                .willReturn(aResponse()
                        .withStatus(200)
                        .withHeader("Content-Type", "application/json")
                        .withBody("{\"result_id\":\"szybki-ocr-id\"}")));

        // 3. STATUS POLLING -> Rejestracja opóźnień w zależności od ID
        registerStatusStubs(server, "wolny-ocr-id", SLOW_DELAY_MS, inProgressJson, doneJson);
        registerStatusStubs(server, "szybki-ocr-id", FAST_DELAY_MS, inProgressJson, doneJson);

        // 4. DOWNLOAD PDF -> Zawsze to samo
        server.stubFor(get(urlPathEqualTo("/api/downloadPdf"))
                .withHeader("Authorization", matching(".*"))
                .willReturn(aResponse()
                        .withStatus(200)
                        .withHeader("Content-Type", "application/json")
                        .withBody(downloadJson)));

        log.info("[LOCAL-STUB] registered WireMock stubs for Fast & Slow processing simulation");
    }

    private static void registerStatusStubs(WireMockServer server, String id, int delayMs, String inProgressJson, String doneJson) {
        String scenarioName = "Status-Scenario-" + id;

        // Pierwszy polling -> zwraca IN PROGRESS + nakłada twardy delay
        server.stubFor(get(urlPathEqualTo("/api/status/" + id))
                .inScenario(scenarioName)
                .whenScenarioStateIs(Scenario.STARTED)
                .willReturn(aResponse()
                        .withStatus(200)
                        .withHeader("Content-Type", "application/json")
                        .withFixedDelay(delayMs)
                        .withBody(inProgressJson))
                .willSetStateTo("InProgress"));

        // Drugi polling -> zwraca SUCCESS + nakłada twardy delay
        server.stubFor(get(urlPathEqualTo("/api/status/" + id))
                .inScenario(scenarioName)
                .whenScenarioStateIs("InProgress")
                .willReturn(aResponse()
                        .withStatus(200)
                        .withHeader("Content-Type", "application/json")
                        .withFixedDelay(delayMs)
                        .withBody(doneJson))
                .willSetStateTo("Done"));

        // Kolejne strzały po udanym procesowaniu (Idempotentność) -> SUCCESS bez opóźnień
        server.stubFor(get(urlPathEqualTo("/api/status/" + id))
                .inScenario(scenarioName)
                .whenScenarioStateIs("Done")
                .willReturn(aResponse()
                        .withStatus(200)
                        .withHeader("Content-Type", "application/json")
                        .withBody(doneJson)));
    }
}
